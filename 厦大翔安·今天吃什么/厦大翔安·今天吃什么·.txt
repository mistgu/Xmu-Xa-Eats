const DATA_KEY = 'xmu_data_v3';
let DATA = loadData();
function loadData() {
  try {
    const s = localStorage.getItem(DATA_KEY);
    if (s) { const d = JSON.parse(s); if (Array.isArray(d) && d.length) return d; }
  } catch (e) {}
  return RAW_DATA;
}
function saveData() { try { localStorage.setItem(DATA_KEY, JSON.stringify(DATA)); } catch (e) {} }
function resetData() {
  localStorage.removeItem(DATA_KEY);
  DATA = RAW_DATA; reinit();
  toast('已恢复默认数据 🍃');
}

function csvField(v) {
  const s = String(v == null ? '' : v);
  if (/[",\n\r;]/.test(s)) return '"' + s.replace(/"/g, '""') + '"';
  return s;
}
function parseCSV(text) {
  const rows = []; let row = []; let cur = ''; let inQ = false;
  const s = String(text || '').replace(/^﻿/, '');
  for (let i = 0; i < s.length; i++) {
    const c = s[i];
    if (inQ) {
      if (c === '"') { if (s[i + 1] === '"') { cur += '"'; i++; } else inQ = false; }
      else cur += c;
    } else {
      if (c === '"') inQ = true;
      else if (c === ',') { row.push(cur); cur = ''; }
      else if (c === '\n' || c === '\r') { if (c === '\r' && s[i + 1] === '\n') i++; row.push(cur); cur = ''; rows.push(row); row = []; }
      else cur += c;
    }
  }
  if (cur !== '' || row.length) { row.push(cur); rows.push(row); }
  return rows.filter(r => r.some(c => c.trim() !== ''));
}

const CSV_HEAD = ['店名','位置','类型','区域','营业时间','外卖','店铺评分','店铺测评人数','店铺标签','店铺简介','菜名','价格','价格显示','菜品评分','推荐人数','招牌','菜品标签','菜品描述'];
function dataToCSV() {
  const lines = [CSV_HEAD.map(csvField).join(',')];
  for (const r of DATA) for (const d of r.dishes) {
    lines.push([r.name, r.zone, r.kind, r.area, r.open, deliveryOf(r) ? '1' : '0', r.rating, r.reviewCount,
      r.tags.join(';'), r.intro, d.name, d.price, d.priceText, d.rating, d.hot,
      d.rec ? '1' : '0', d.tags.join(';'), d.desc].map(csvField).join(','));
  }
  return lines.join('\r\n');
}
function csvToData(text) {
  const rows = parseCSV(text);
  if (!rows.length) return null;
  let dataRows = rows;
  if (rows[0][0] && rows[0][0].trim() === '店名') dataRows = rows.slice(1);
  const map = new Map();
  for (const row of dataRows) {
    const [name, zone, kind, area, open, dlv, rating, reviewCount, storeTags, intro,
           dish, price, priceText, dishRating, hot, rec, dishTags, dishDesc] = row;
    const dlvVal = (dlv === '1' || dlv === '是' || dlv === 'true') ? true
      : (dlv === '0' || dlv === '否' || dlv === 'false') ? false : undefined;
    if (!name || !dish) continue;
    if (!map.has(name)) {
      const obj = {
        name, zone: zone || '校外', kind: kind || '餐厅', area: area || '', open: open || '10:00-22:00',
        rating: parseFloat(rating) || 4.0, reviewCount: parseInt(reviewCount) || 0,
        tags: (storeTags || '').split(/[;；]/).filter(Boolean), intro: intro || '', dishes: [],
      };
      if (dlvVal !== undefined) obj.delivery = dlvVal;
      map.set(name, obj);
    }
    const r = map.get(name);
    r.dishes.push({
      name: dish, price: parseFloat(price) || 0, priceText: priceText || ('¥' + (price || '?')),
      rating: parseFloat(dishRating) || 4.0, hot: parseInt(hot) || 0,
      rec: rec === '1' || rec === '是' || rec === 'true', tags: (dishTags || '').split(/[;；]/).filter(Boolean), desc: dishDesc || '',
    });
  }
  if (!map.size) return null;
  const merged = [];
  const seen = new Set();
  for (const r of map.values()) {
    const existing = DATA.find(x => x.name === r.name);
    if (existing) {
      const m = { ...existing, zone: r.zone, kind: r.kind, area: r.area, open: r.open, rating: r.rating,
        reviewCount: r.reviewCount, tags: r.tags, intro: r.intro, dishes: r.dishes };
      if (r.delivery !== undefined) m.delivery = r.delivery;
      merged.push(m);
    } else {
      merged.push({ ...r, id: r.name, emoji: KIND_EMOJI[r.kind] || '🍴', reviews: [] });
    }
    seen.add(r.name);
  }
  for (const r of DATA) if (!seen.has(r.name)) merged.push(r);
  return merged;
}

function exportCSV() { downloadText('﻿' + dataToCSV(), '厦大翔安美食数据.csv'); toast('CSV 已导出 📤'); }
function downloadCSVTemplate() {
  const example = ['示例店','校外','餐厅','沙美学生街','10:00-22:00','1','4.5','88','人气;平价','这是一家示例店铺','示例菜','15','¥15','4.5','9','1','米饭;辣','示例描述'].map(csvField).join(',');
  downloadText('﻿' + CSV_HEAD.map(csvField).join(',') + '\r\n' + example, '厦大翔安美食导入模板.csv');
  toast('模板已下载 📄');
}
function handleCsvFile(input) {
  const file = input.files && input.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = () => {
    const nd = csvToData(reader.result);
    if (!nd) { toast('❌ CSV 解析失败，请用模板格式'); input.value = ''; return; }
    DATA = nd; saveData(); reinit();
    toast('✅ 已导入 ' + nd.length + ' 家店铺');
    input.value = '';
  };
  reader.readAsText(file, 'utf-8');
}
function downloadText(text, filename) {
  const blob = new Blob([text], { type: 'text/csv;charset=utf-8' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = filename; a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 1000);
}

function seedOfDay(offset) {
  const n = new Date();
  return n.getFullYear() * 10000 + (n.getMonth() + 1) * 100 + n.getDate() + (offset || 0);
}
function dailyPick(offset) {
  const order = ALL_DISHES.slice().sort((a, b) => a.id.localeCompare(b.id));
  if (!order.length) return null;
  return order[seedOfDay(offset) % order.length];
}
function renderDaily() {
  const n = new Date();
  const wk = ['日','一','二','三','四','五','六'][n.getDay()];
  document.getElementById('dailyDate').textContent = `${n.getMonth() + 1}月${n.getDate()}日 · 周${wk}`;
  const pick = dailyPick(state.dailyOffset);

function canvasBlob(canvas) { return new Promise(res => canvas.toBlob(res, 'image/png')); }
async function downloadCanvas(canvas, filename) {
  const blob = await canvasBlob(canvas);
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = filename; a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 1200);
}
async function shareOrDownload(canvas, filename) {
  const blob = await canvasBlob(canvas);
  if (navigator.canShare && navigator.canShare({ files: [new File([blob], filename, { type: 'image/png' })] })) {
    try { await navigator.share({ files: [new File([blob], filename, { type: 'image/png' })], title: filename }); return; }
    catch (e) { if (e && e.name === 'AbortError') return; }
  }
  await downloadCanvas(canvas, filename);
  toast('图片已保存 📤');
}

function drawShareCard({ title, subtitle, rows, footer, accent = '#5b9168' }) {
  const W = 640, pad = 44, headH = 120, lineH = 50, footH = 70;
  const bodyTop = headH + 26;
  const H = headH + 26 + rows.length * lineH + 18 + footH;
  const scale = 2;
  const canvas = document.createElement('canvas');
  canvas.width = W * scale; canvas.height = H * scale;
  const ctx = canvas.getContext('2d');
  ctx.scale(scale, scale);

  const bg = ctx.createLinearGradient(0, 0, 0, H);
  bg.addColorStop(0, '#f3f8f4'); bg.addColorStop(1, '#e8f1eb');
  ctx.fillStyle = bg; ctx.fillRect(0, 0, W, H);

  const hg = ctx.createLinearGradient(0, 0, W, 0);
  hg.addColorStop(0, '#8fc09d'); hg.addColorStop(1, accent);
  ctx.fillStyle = hg; ctx.fillRect(0, 0, W, headH);

  ctx.textBaseline = 'middle'; ctx.textAlign = 'left';
  ctx.fillStyle = '#fff'; ctx.font = '700 30px "PingFang SC","Microsoft YaHei",sans-serif';
  ctx.fillText(trunc(ctx, title, W - pad * 2), pad, 44);
  if (subtitle) {
    ctx.fillStyle = 'rgba(255,255,255,.92)'; ctx.font = '400 15px "PingFang SC","Microsoft YaHei",sans-serif';
    ctx.fillText(trunc(ctx, subtitle, W - pad * 2), pad, 78);
  }

  rows.forEach((row, i) => {
    const y = bodyTop + i * lineH + lineH / 2;
    if (row.highlight) {
      ctx.fillStyle = 'rgba(91,145,104,.10)';
      ctx.beginPath();
      if (ctx.roundRect) ctx.roundRect(pad - 12, y - lineH / 2 + 6, W - (pad - 12) * 2, lineH - 6, 12);
      else ctx.rect(pad - 12, y - lineH / 2 + 6, W - (pad - 12) * 2, lineH - 6);
      ctx.fill();
    }
    ctx.fillStyle = '#2f3b31'; ctx.font = '600 17px "PingFang SC","Microsoft YaHei",sans-serif';
    ctx.textAlign = 'left';
    ctx.fillText(trunc(ctx, row.left, W - pad * 2 - 120), pad, y - (row.sub ? 8 : 0));
    ctx.textAlign = 'right'; ctx.fillStyle = '#e79a63'; ctx.font = '700 17px "PingFang SC","Microsoft YaHei",sans-serif';
    ctx.fillText(row.right || '', W - pad, y - (row.sub ? 8 : 0));
    ctx.textAlign = 'left';
    if (row.sub) {
      ctx.fillStyle = '#82908a'; ctx.font = '400 12px "PingFang SC","Microsoft YaHei",sans-serif';
      ctx.fillText(trunc(ctx, row.sub, W - pad * 2 - 120), pad, y + 12);
    }
  });

  ctx.fillStyle = '#82908a'; ctx.font = '400 13px "PingFang SC","Microsoft YaHei",sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText(footer, W / 2, H - 26);
  return canvas;
}
function trunc(ctx, text, maxW) {
  if (ctx.measureText(text).width <= maxW) return text;
  let t = text;
  while (t.length > 1 && ctx.measureText(t + '…').width > maxW) t = t.slice(0, -1);
  return t + '…';
}

/* ============================================================
 * 筛选逻辑
 * ============================================================ */
function matchLoc(zone, f) { return f === '全部' || zone === f; }
function matchBudget(price, f) {
  if (f === '不限') return true;
  if (f === '0-15') return price <= 15;
  if (f === '15-25') return price > 15 && price <= 25;
  if (f === '25-') return price > 25;
  return true;
}
function matchTaste(tags, f) {
  if (f === '都行') return true;
  if (f === '辣') return tags.includes('辣');
  if (f === '清淡') return tags.includes('清淡');
  if (f === '甜') return tags.includes('甜');
  return true;
}
function matchMode(r, f) { return f === '都行' || (f === '外卖' ? deliveryOf(r) : true); }
function matchCat(item, cat) {
  if (cat === '全部') return true;
  if (cat === '校内' || cat === '校外') return item.r.zone === cat;
  return item.d.tags.includes(cat);
}

function renderSearchResults(el) {
  const q = state.search.trim().toLowerCase();
  const hits = ALL_DISHES.filter(x =>
    x.d.name.toLowerCase().includes(q) ||
    x.r.name.toLowerCase().includes(q) ||
    x.d.desc.toLowerCase().includes(q) ||
    x.d.tags.some(t => t.includes(q)) ||
    x.r.kind.includes(q) || x.r.area.includes(q)

  if (!hits.length) { el.innerHTML = emptyHtml(`没有找到和「${q}」相关的内容`, '🔍'); return; }
  el.innerHTML = `<div class="dish-flat">` + hits.map(x => dishItemHtml(x)).join('') + `</div>`;
}

const REASONS = {
  '辣': ['今天适合来点刺激的，辣到过瘾！', '无辣不欢星人，冲就完事！', '来点辣的，快乐加倍～'],
  '清淡': ['清淡一点，肠胃更舒服～', '天气正好，来点清爽的！', '今天宜清淡，养养胃。'],
  '甜': ['生活这么苦，来点甜的吧！', '甜甜的，治愈一整天！', '今天宜补充糖分～'],
  default: ['闭眼选它，绝不踩雷！', '人气之选，跟着大家吃没错！', '好评如潮，就它了！', '缘分让我们相遇，吃它！'],
};
function pickReason(tags) {
  for (const k of ['辣', '清淡', '甜']) if (tags.includes(k)) {
    const arr = REASONS[k];
    return arr[Math.floor(Math.random() * arr.length)];
  }
  const arr = REASONS.default;
  return arr[Math.floor(Math.random() * arr.length)];
}

let animTimer = null;
function decide() {
  const pool = ALL_DISHES.filter(x =>
    matchLoc(x.r.zone, state.fLoc) &&
    matchBudget(x.d.price, state.fBudget) &&
    matchTaste(x.d.tags, state.fTaste) &&
    matchMode(x.r, state.fMode)
  );
  if (!pool.length) { toast('这个条件下暂时没有菜品哦，换个筛选试试 🍃'); return; }

  // 轻微加权：招牌/人气高的更可能被选中
  const weights = pool.map(x => (x.d.rec ? 3 : 1) * (x.d.hot || 1));
  const total = weights.reduce((a, b) => a + b, 0);
  let rnd = Math.random() * total;
  let idx = 0;
  for (let i = 0; i < pool.length; i++) { rnd -= weights[i]; if (rnd <= 0) { idx = i; break; } }
  const pick = pool[idx];

  // 滚动动画：先快速切换名字
  const box = document.getElementById('resultBox');
  const nameEl = document.getElementById('resDish');
  box.classList.remove('show');
  void box.offsetWidth;
  box.classList.add('show');
  if (animTimer) clearInterval(animTimer);
  let count = 0;
  nameEl.textContent = '🍳 想吃什么好呢…';
  animTimer = setInterval(() => {
    const tmp = pool[Math.floor(Math.random() * pool.length)];
    nameEl.textContent = tmp.d.name;
    count++;
    if (count > 10) {
      clearInterval(animTimer);
      fillResult(pick);
    }
  }, 70);
}

function fillResult(pick) {
  state.lastPickId = pick.id;
  document.getElementById('resDish').textContent = pick.d.name;
  document.getElementById('resPrice').textContent = pick.d.priceText;
  document.getElementById('resRest').innerHTML = `📍 <b>${pick.r.name}</b> · ${pick.r.area}`;
  document.getElementById('resStars').innerHTML =
    `${'★'.repeat(Math.round(pick.d.rating))}${'☆'.repeat(5 - Math.round(pick.d.rating))}<span>${pick.d.rating} 分 · ${pick.d.hot}人推荐</span>`;
  document.getElementById('resReason').textContent = '💡 ' + pickReason(pick.d.tags);
  document.getElementById('resDesc').textContent = pick.d.desc;
  document.getElementById('resFavBtn').textContent = isFav(pick.id) ? '💚 已收藏' : '🤍 收藏';
}

function toggleFavFromResult() {
  if (!state.lastPickId) return;
  const on = toggleFav(state.lastPickId);
  document.getElementById('resFavBtn').textContent = on ? '💚 已收藏' : '🤍 收藏';
  if (state.tab === 'fav') renderFav();
  toast(on ? '已收藏 💚' : '已取消收藏');
}

function closeSheet() {
  document.getElementById('sheet').classList.remove('show');

/* ============================================================
 * 交互
 * ============================================================ */
function toggleFavAndRender(id, btn) {
  const on = toggleFav(id);
  if (btn) btn.textContent = on ? '💚' : '🤍';
  if (state.tab === 'fav') renderFav();
  if (state.lastPickId === id) {
    document.getElementById('resFavBtn').textContent = on ? '💚 已收藏' : '🤍 收藏';
  }
}

function setCat(cat) { state.cat = cat; renderTabs(); renderBrowse(); }

function onSearch(v) {
  state.search = v;
  document.getElementById('searchClear').style.display = v ? 'inline' : 'none';
  renderBrowse();
}
function clearSearch() {
  state.search = '';
  document.getElementById('searchInput').value = '';
  document.getElementById('searchClear').style.display = 'none';
  renderBrowse();
}

function switchTab(tab) {
  state.tab = tab;
  document.querySelectorAll('.nav-item').forEach(n => n.classList.toggle('active', n.dataset.tab === tab));
  document.getElementById('view-home').style.display = tab === 'home' ? 'block' : 'none';
  document.getElementById('view-rank').style.display = tab === 'rank' ? 'block' : 'none';
  document.getElementById('view-fav').style.display = tab === 'fav' ? 'block' : 'none';
  if (tab === 'rank') renderRank();
  if (tab === 'fav') renderFav();
  if (tab === 'home') renderBrowse();
  window.scrollTo({ top: 0 });
}

/* 筛选 chips 绑定 */
function bindFilter(id, key) {
  const wrap = document.getElementById(id);
  wrap.addEventListener('click', e => {
    const chip = e.target.closest('.chip');
    if (!chip) return;
    wrap.querySelectorAll('.chip').forEach(c => c.classList.remove('active'));
    chip.classList.add('active');
    state[key] = chip.dataset.v;
  });
}

/* toast */
let toastTimer = null;
function toast(msg) {
  let t = document.getElementById('toast');
  if (!t) {
    t = document.createElement('div');
    t.id = 'toast';
    t.style.cssText = 'position:fixed;left:50%;top:50%;transform:translate(-50%,-50%);background:rgba(47,59,49,.92);color:#fff;padding:12px 22px;border-radius:14px;font-size:14px;z-index:300;max-width:80vw;text-align:center;transition:opacity .3s;';
    document.body.appendChild(t);
  }
  t.textContent = msg;
  t.style.opacity = '1';
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => { t.style.opacity = '0'; }, 1600);
}
